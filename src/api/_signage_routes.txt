
// -- Signage Materials --

app.get("/api/signage-material-categories", authMiddleware, async (_req, res) => {
  try {
    const categories = await db.select().from(signageMaterialCategories).orderBy(signageMaterialCategories.sortOrder);
    res.json(categories);
  } catch (error) { console.error(error); res.status(500).json({ error: "Failed" }); }
});

app.post("/api/signage-material-categories", authMiddleware, async (req, res) => {
  try {
    const user = (req as any).user;
    if (!isAdminUser(user)) return res.status(403).json({ error: "Admin access required" });
    const { name, slug, description, sortOrder } = req.body;
    const [category] = await db.insert(signageMaterialCategories).values({ name, slug, description, sortOrder: sortOrder || 0 }).returning();
    res.json(category);
  } catch (error) { console.error(error); res.status(500).json({ error: "Failed" }); }
});

app.put("/api/signage-material-categories/:id", authMiddleware, async (req, res) => {
  try {
    const user = (req as any).user;
    if (!isAdminUser(user)) return res.status(403).json({ error: "Admin access required" });
    const { name, slug, description, sortOrder, isActive } = req.body;
    const [updated] = await db.update(signageMaterialCategories).set({ name, slug, description, sortOrder, isActive }).where(eq(signageMaterialCategories.id, req.params.id)).returning();
    if (!updated) return res.status(404).json({ error: "Not found" });
    res.json(updated);
  } catch (error) { console.error(error); res.status(500).json({ error: "Failed" }); }
});

app.delete("/api/signage-material-categories/:id", authMiddleware, async (req, res) => {
  try {
    const user = (req as any).user;
    if (!isAdminUser(user)) return res.status(403).json({ error: "Admin access required" });
    await db.delete(signageMaterialCategories).where(eq(signageMaterialCategories.id, req.params.id));
    res.json({ success: true });
  } catch (error) { console.error(error); res.status(500).json({ error: "Failed" }); }
});

app.get("/api/signage-configs/:id", async (req, res) => {
  try {
    const [config] = await db.select().from(signageProductConfigs).where(eq(signageProductConfigs.id, req.params.id));
    if (!config) return res.status(404).json({ error: "Not found" });
    const configMaterials = await db.select().from(signageConfigMaterials).innerJoin(signageMaterials, eq(signageConfigMaterials.materialId, signageMaterials.id)).where(eq(signageConfigMaterials.configId, req.params.id));
    res.json({ ...config, materials: configMaterials });
  } catch (error) { console.error(error); res.status(500).json({ error: "Failed" }); }
});
